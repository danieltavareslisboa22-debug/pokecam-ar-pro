  <script>
    // Abertura com cursor dourado
    const textoCompleto = "Bem‑vindo ao Mundo das Criaturas...";
    const elIntro = document.getElementById("textoIntro");
    let indice = 0;
    function digitar() {
      if(indice < textoCompleto.length) {
        elIntro.textContent += textoCompleto[indice];
        indice++;
        setTimeout(digitar, 80);
      } else {
        elIntro.innerHTML += '<span class="blink">|</span>';
        setTimeout(()=>{
          document.getElementById("intro").classList.add("fade‑out");
        }, 1800);
      }
    }
    digitar();

    // Navegação entre seções
    const botoesNav = document.querySelectorAll("nav button");
    const secoes = document.querySelectorAll(".secao");
    botoesNav.forEach(botao => {
      botao.addEventListener("click", ()=>{
        const alvo = botao.dataset.secao;
        secoes.forEach(s => s.classList.remove("ativa"));
        document.getElementById(alvo).classList.add("ativa");
      });
    });

    // Câmera, captura de foto e nota aleatória
    let streamCam = null;
    const video = document.getElementById("video");
    const canvas = document.getElementById("canvas");
    const ctx = canvas.getContext("2d");
    const fotoCapturada = document.getElementById("fotoCapturada");
    const btnAbrirCam = document.getElementById("btnAbrirCam");
    const btnTirarFoto = document.getElementById("btnTirarFoto");
    const resultadoNota = document.getElementById("resultadoNota");
    const historico = document.getElementById("historico");
    let registros = JSON.parse(localStorage.getItem("registrosCriaturas")) || [];

    function atualizarHistorico() {
      historico.innerHTML = "";
      registros.forEach((reg, i)=>{
        historico.innerHTML += `<div class="card">Registro ${i+1} — Nota: ${reg.nota}/100 — Data: ${reg.data}</div>`;
      });
    }
    atualizarHistorico();

    btnAbrirCam.addEventListener("click", async ()=>{
      try {
        streamCam = await navigator.mediaDevices.getUserMedia({video:{facingMode:"environment"}});
        video.srcObject = streamCam;
        btnAbrirCam.style.display = "none";
        btnTirarFoto.style.display = "block";
      } catch(erro) {
        alert("Não foi possível abrir a câmera: permissão negada ou não disponível.");
      }
    });

    btnTirarFoto.addEventListener("click", ()=>{
      // Salva a foto do vídeo para o site
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      ctx.drawImage(video, 0, 0);
      const imagemUrl = canvas.toDataURL("image/png");
      fotoCapturada.src = imagemUrl;
      fotoCapturada.style.display = "block";

      // Nota aleatória de 0 a 100
      const notaAleatoria = Math.floor(Math.random()*101);
      resultadoNota.textContent = `⭐ Nota da criatura: ${notaAleatoria}/100`;
      const agora = new Date().toLocaleString("pt‑BR");
      registros.push({nota:notaAleatoria, data:agora, imagem:imagemUrl});
      localStorage.setItem("registrosCriaturas", JSON.stringify(registros));
      atualizarHistorico();
    });

    // Partículas simples de fundo
    const containerParticulas = document.getElementById("particulas");
    for(let i=0; i<35; i++) {
      const p = document.createElement("div");
      p.style.position = "absolute";
      p.style.width = Math.random()*6+2+"px";
      p.style.height = p.style.width;
      p.style.background = "#ffd700";
      p.style.borderRadius = "50%";
      p.style.opacity = Math.random()*0.4+0.1;
      p.style.left = Math.random()*100+"%";
      p.style.top = Math.random()*100+"%";
      containerParticulas.appendChild(p);
    }
  </script>
